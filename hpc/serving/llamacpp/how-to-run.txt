# Serve a model with llama.cpp on ONE GPU.
# For the multi-GPU version, submit run.sh instead.
#
# llama.cpp does NOT read a Hugging Face model folder -- it needs a single .gguf
# file. Convert your fine-tuned "merged" folder once, on a login node:
#
#   git clone https://github.com/ggml-org/llama.cpp
#   pip install -r llama.cpp/requirements.txt
#   python llama.cpp/convert_hf_to_gguf.py /path/to/models/merged \
#       --outfile /path/to/models/qwen-sql-f16.gguf --outtype f16
#
# Smaller and faster, at a small quality cost (optional):
#   apptainer exec llamacpp-b10254-cuda13.3.sif \
#     llama-quantize /path/to/models/qwen-sql-f16.gguf \
#                    /path/to/models/qwen-sql-q4.gguf Q4_K_M


# 1. get an interactive GPU node
salloc --gres=gpu:1 --cpus-per-task=8 --time=02:00:00


# 2. start the server -- port 8080
apptainer run --nv \
  --bind /path/to/models:/model \
  llamacpp-b10254-cuda13.3.sif serve \
  -m /model/qwen-sql-f16.gguf \
  -ngl 99

# -ngl 99  = put all layers on the GPU. Lower it if the model doesn't fit.
# -c 4096  = context length, if you need more than the default.


# 3. check it, from the same node
curl http://localhost:8080/v1/models

curl http://localhost:8080/v1/chat/completions \
  -H 'Content-Type: application/json' \
  -d '{"messages":[{"role":"user","content":"CREATE TABLE head (age INTEGER)\n\nHow many heads are older than 56?"}]}'


# 4. use it from your laptop -- tunnel the port, then open http://localhost:8080
ssh -L 8080:<compute-node>:8080 <you>@<login-node>
