# Serve a model with SGLang on ONE GPU.
# For 2 nodes x 2 GPUs, submit run.sh instead.
#
# You need: the .sif in this folder, and a model directory (your fine-tuned
# "merged" folder, or a download from ../download-model.py).


# 1. get an interactive GPU node
salloc --gres=gpu:1 --cpus-per-task=8 --time=02:00:00


# 2. start the server -- port 30000
apptainer run --nv \
  --bind /path/to/models:/model \
  sglang-0.5.17-cuda13.0.sif serve

# it serves whatever sits at /model. To point at a subfolder:
#   --env MODEL=/model/qwen-sql
# useful extras go after `serve`, e.g.:
#   serve --context-length 4096 --mem-fraction-static 0.85


# 3. check it, from the same node (wait for "The server is fired up")
curl http://localhost:30000/v1/models

curl http://localhost:30000/v1/chat/completions \
  -H 'Content-Type: application/json' \
  -d '{"model":"/model","messages":[{"role":"user","content":"CREATE TABLE head (age INTEGER)\n\nHow many heads are older than 56?"}]}'


# 4. use it from your laptop -- tunnel the port, then open http://localhost:30000/docs
ssh -L 30000:<compute-node>:30000 <you>@<login-node>
